# soundctl Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Menu bar app that sends each app's audio to a chosen output device at a per-app volume (e.g. Aside → speakers, Discord → earphones).

**Architecture:** Per routed app, a muted Core Audio process tap feeds a private aggregate device (target output device + tap); an IOProc copies tap samples × gain to the output. Rules are stored per bundle ID and re-applied when apps/devices change. Spec: `docs/superpowers/specs/2026-11-01-soundctl-design.md`.

**Tech Stack:** Swift 6 toolchain (language mode 5), SwiftPM, SwiftUI `MenuBarExtra`, CoreAudio (`AudioHardwareCreateProcessTap`, `AudioHardwareCreateAggregateDevice`), `Synchronization.Atomic`, `ServiceManagement`, XCTest.

## Global Constraints

- Platform: macOS 26.0 (`platforms: [.macOS("26.0")]`, `LSMinimumSystemVersion` 26.0).
- No third-party dependencies, no virtual audio driver.
- Bundle ID `com.gsw226.soundctl`, `LSUIElement = true`, `NSAudioCaptureUsageDescription` present.
- Apps on system default output at 100% get no tap.
- UI strings in Korean: "시스템 기본", "오디오 권한 필요", "설정 열기", "로그인 시 실행", "종료".

import Foundation

/// What the user chose for one app. `deviceUID == nil` means "system default output".
struct Rule: Codable, Equatable {
    var deviceUID: String?
    var deviceName: String?
    var volume: Float = 1

    /// Default device at 100% needs no tap at all.
    var isDefault: Bool { deviceUID == nil && volume >= 1 }
}

/// Rules keyed by app bundle ID, persisted as JSON in UserDefaults.
struct RouteStore {
    var defaults: UserDefaults = .standard
    private let key = "rules"

    func load() -> [String: Rule] {
        guard let data = defaults.data(forKey: key),
              let rules = try? JSONDecoder().decode([String: Rule].self, from: data) else { return [:] }
        return rules
    }

    func save(_ rules: [String: Rule]) {
        defaults.set(try? JSONEncoder().encode(rules), forKey: key)
    }
}

// MARK: - Core Audio property helpers

func caAddress(_ selector: AudioObjectPropertySelector,
               _ scope: AudioObjectPropertyScope = kAudioObjectPropertyScopeGlobal) -> AudioObjectPropertyAddress {
    AudioObjectPropertyAddress(mSelector: selector, mScope: scope, mElement: kAudioObjectPropertyElementMain)
}

func caIDs(_ object: AudioObjectID, _ selector: AudioObjectPropertySelector,
           _ scope: AudioObjectPropertyScope = kAudioObjectPropertyScopeGlobal) -> [AudioObjectID] {
    var address = caAddress(selector, scope)
    var size: UInt32 = 0
    guard AudioObjectGetPropertyDataSize(object, &address, 0, nil, &size) == noErr, size > 0 else { return [] }
    var ids = [AudioObjectID](repeating: 0, count: Int(size) / MemoryLayout<AudioObjectID>.size)
    guard AudioObjectGetPropertyData(object, &address, 0, nil, &size, &ids) == noErr else { return [] }
    return ids
}

func caValue<T: BitwiseCopyable>(_ object: AudioObjectID, _ selector: AudioObjectPropertySelector, _ initial: T) -> T? {
    var address = caAddress(selector)
    var size = UInt32(MemoryLayout<T>.size)
    var value = initial
    guard AudioObjectGetPropertyData(object, &address, 0, nil, &size, &value) == noErr else { return nil }
    return value
}

func caString(_ object: AudioObjectID, _ selector: AudioObjectPropertySelector) -> String? {
    var address = caAddress(selector)
    var size = UInt32(MemoryLayout<Unmanaged<CFString>?>.size)
    var value: Unmanaged<CFString>?
    guard AudioObjectGetPropertyData(object, &address, 0, nil, &size, &value) == noErr else { return nil }
    return value?.takeRetainedValue() as String?
}

let systemObject = AudioObjectID(kAudioObjectSystemObject)

/// Calls `handler` on the main queue whenever a system-object property changes.
func observeSystem(_ selector: AudioObjectPropertySelector, _ handler: @escaping () -> Void) {
    var address = caAddress(selector)
    AudioObjectAddPropertyListenerBlock(systemObject, &address, .main) { _, _ in handler() }
}

// MARK: - Apps

/// An app with one or more Core Audio process objects (main process + helpers).
struct AudioApp: Identifiable {
    let bundleID: String
    let name: String
    let icon: NSImage?
    var processes: [AudioObjectID]
    var isPlaying: Bool
    var id: String { bundleID }
}

/// `com.hnc.Discord.helper.Renderer` → `com.hnc.Discord` when `com.hnc.Discord` is a running app.
/// Longest matching app wins; unmatched IDs map to themselves.
func parentBundleID(of id: String, apps: Set<String>) -> String {
    apps.filter { id == $0 || id.hasPrefix($0 + ".") }.max { $0.count < $1.count } ?? id
}

/// Running apps that own Core Audio process objects, helpers folded into their parent app.
func listAudioApps() -> [AudioApp] {
    let running = NSWorkspace.shared.runningApplications.filter { $0.activationPolicy == .regular }
    let byID = Dictionary(running.compactMap { app in app.bundleIdentifier.map { ($0, app) } },
                          uniquingKeysWith: { a, _ in a })
    let appIDs = Set(byID.keys)
    let me = Bundle.main.bundleIdentifier

    var apps: [String: AudioApp] = [:]
    for process in caIDs(systemObject, kAudioHardwarePropertyProcessObjectList) {
        guard let bundle = caString(process, kAudioProcessPropertyBundleID), !bundle.isEmpty else { continue }
        let parent = parentBundleID(of: bundle, apps: appIDs)
        guard parent != me, let app = byID[parent] else { continue }
        let playing = (caValue(process, kAudioProcessPropertyIsRunningOutput, UInt32(0)) ?? 0) != 0
        apps[parent, default: AudioApp(bundleID: parent, name: app.localizedName ?? parent,
                                       icon: app.icon, processes: [], isPlaying: false)]
            .processes.append(process)
        if playing { apps[parent]?.isPlaying = true }
    }
    return apps.values.sorted { $0.name.localizedCaseInsensitiveCompare($1.name) == .orderedAscending }
}

// MARK: - Output devices

struct OutputDevice: Identifiable, Hashable {
    let uid: String
    let name: String
    var id: String { uid }
}

/// Physical/virtual output devices, excluding aggregates (ours included).
func listOutputDevices() -> [OutputDevice] {
    caIDs(systemObject, kAudioHardwarePropertyDevices).compactMap { device in
        guard !caIDs(device, kAudioDevicePropertyStreams, kAudioObjectPropertyScopeOutput).isEmpty,
              caValue(device, kAudioDevicePropertyTransportType, UInt32(0)) != kAudioDeviceTransportTypeAggregate,
              let uid = caString(device, kAudioDevicePropertyDeviceUID),
              let name = caString(device, kAudioObjectPropertyName) else { return nil }
        return OutputDevice(uid: uid, name: name)
    }
}

func defaultOutputUID() -> String? {
    guard let device = caValue(systemObject, kAudioHardwarePropertyDefaultOutputDevice, AudioObjectID(0)),
          device != kAudioObjectUnknown else { return nil }
    return caString(device, kAudioDevicePropertyDeviceUID)
}

func deviceID(forUID uid: String) -> AudioObjectID? {
    caIDs(systemObject, kAudioHardwarePropertyDevices).first { caString($0, kAudioDevicePropertyDeviceUID) == uid }
}

import CoreAudio
import Foundation
import Synchronization

struct RouteError: Error, CustomStringConvertible {
    let step: String
    let status: OSStatus
    var description: String { "\(step) failed (\(status))" }
}

/// Gain shared with the realtime IO thread. Stored as Float bits in an atomic.
final class Gain: Sendable {
    private let bits: Atomic<UInt32>
    init(_ value: Float) { bits = Atomic(value.bitPattern) }
    var value: Float {
        get { Float(bitPattern: bits.load(ordering: .relaxed)) }
        set { bits.store(newValue.bitPattern, ordering: .relaxed) }
    }
}

/// Routes one app's processes to one output device at a gain:
/// muted process tap → private aggregate device (output device + tap) → IOProc copies tap × gain to output.
final class AppRoute {
    let processes: [AudioObjectID]
    let deviceUID: String
    private let gain: Gain
    private var tapID = AudioObjectID(kAudioObjectUnknown)
    private var aggregateID = AudioObjectID(kAudioObjectUnknown)
    private var procID: AudioDeviceIOProcID?

    var volume: Float {
        get { gain.value }
        set { gain.value = newValue }
    }

    init(processes: [AudioObjectID], deviceUID: String, volume: Float) throws {
        self.processes = processes
        self.deviceUID = deviceUID
        self.gain = Gain(volume)
        do { try start() } catch { stop(); throw error }
    }

    deinit { stop() }

    private func start() throws {
        let tap = CATapDescription(stereoMixdownOfProcesses: processes)
        tap.uuid = UUID()
        tap.muteBehavior = .muted
        tap.isPrivate = true
        try check("create tap", AudioHardwareCreateProcessTap(tap, &tapID))

        let description: [String: Any] = [
            kAudioAggregateDeviceNameKey: "soundctl",
            kAudioAggregateDeviceUIDKey: UUID().uuidString,
            kAudioAggregateDeviceMainSubDeviceKey: deviceUID,
            kAudioAggregateDeviceIsPrivateKey: true,
            kAudioAggregateDeviceIsStackedKey: false,
            kAudioAggregateDeviceTapAutoStartKey: true,
            kAudioAggregateDeviceSubDeviceListKey: [[kAudioSubDeviceUIDKey: deviceUID]],
            kAudioAggregateDeviceTapListKey: [[kAudioSubTapUIDKey: tap.uuid.uuidString,
                                               kAudioSubTapDriftCompensationKey: true]],
        ]
        try check("create aggregate", AudioHardwareCreateAggregateDevice(description as CFDictionary, &aggregateID))

        // Input buffers = the output device's own input streams (e.g. a USB headset mic), then the tap.
        let tapBuffer = deviceID(forUID: deviceUID)
            .map { caIDs($0, kAudioDevicePropertyStreams, kAudioObjectPropertyScopeInput).count } ?? 0
        let gain = self.gain
        try check("create IOProc", AudioDeviceCreateIOProcIDWithBlock(&procID, aggregateID, nil) { _, input, _, output, _ in
            render(input: input, output: output, tapBuffer: tapBuffer, gain: gain.value)
        })
        try check("start", AudioDeviceStart(aggregateID, procID))
    }

    func stop() {
        if let procID {
            AudioDeviceStop(aggregateID, procID)
            AudioDeviceDestroyIOProcID(aggregateID, procID)
            self.procID = nil
        }
        if aggregateID != kAudioObjectUnknown {
            AudioHardwareDestroyAggregateDevice(aggregateID)
            aggregateID = AudioObjectID(kAudioObjectUnknown)
        }
        if tapID != kAudioObjectUnknown {
            AudioHardwareDestroyProcessTap(tapID)
            tapID = AudioObjectID(kAudioObjectUnknown)
        }
    }

    private func check(_ step: String, _ status: OSStatus) throws {
        guard status == noErr else { throw RouteError(step: step, status: status) }
    }
}

/// Copies the interleaved stereo Float32 tap buffer into every output buffer, scaled by `gain`.
/// Output channel c takes tap channel min(c, 1); a mono output gets the left channel.
// ponytail: assumes Float32 on both sides (HAL IOProc canonical format); convert if a device ever reports otherwise.
func render(input: UnsafePointer<AudioBufferList>, output: UnsafeMutablePointer<AudioBufferList>,
            tapBuffer: Int, gain: Float) {
    let inputs = UnsafeMutableAudioBufferListPointer(UnsafeMutablePointer(mutating: input))
    let outputs = UnsafeMutableAudioBufferListPointer(output)
    let source = tapBuffer < inputs.count ? inputs[tapBuffer] : nil

    for out in outputs {
        guard let dst = out.mData?.assumingMemoryBound(to: Float.self) else { continue }
        let outChannels = Int(max(out.mNumberChannels, 1))
        let outSamples = Int(out.mDataByteSize) / MemoryLayout<Float>.size
        guard let source, let src = source.mData?.assumingMemoryBound(to: Float.self) else {
            dst.update(repeating: 0, count: outSamples)
            continue
        }
        let inChannels = Int(max(source.mNumberChannels, 1))
        let frames = min(outSamples / outChannels, Int(source.mDataByteSize) / MemoryLayout<Float>.size / inChannels)
        for frame in 0..<frames {
            for channel in 0..<outChannels {
                dst[frame * outChannels + channel] = src[frame * inChannels + min(channel, inChannels - 1)] * gain
            }
        }
        (dst + frames * outChannels).update(repeating: 0, count: outSamples - frames * outChannels)
    }
}

import AppKit
import CoreAudio
import Observation
import ServiceManagement

/// Owns rules and live routes; keeps routes in sync with running apps and connected devices.
@MainActor @Observable
final class Model {
    private(set) var apps: [AudioApp] = []
    private(set) var devices: [OutputDevice] = []
    private(set) var rules: [String: Rule]
    private(set) var failed: [String: String] = [:]
    private(set) var permissionDenied = false
    private(set) var launchAtLogin = SMAppService.mainApp.status == .enabled

    @ObservationIgnored private var routes: [String: AppRoute] = [:]
    @ObservationIgnored private let store = RouteStore()

    init() {
        rules = store.load()
        for selector in [kAudioHardwarePropertyProcessObjectList,
                         kAudioHardwarePropertyDevices,
                         kAudioHardwarePropertyDefaultOutputDevice] {
            observeSystem(selector) { [weak self] in MainActor.assumeIsolated { self?.refresh() } }
        }
        refresh()
    }

    func rule(for bundleID: String) -> Rule { rules[bundleID] ?? Rule() }

    func refresh() {
        devices = listOutputDevices()
        let running = listAudioApps()
        apps = running.filter { $0.isPlaying || rules[$0.bundleID] != nil }
        permissionDenied = audioCapturePermission() == .denied
        let byID = Dictionary(running.map { ($0.bundleID, $0) }, uniquingKeysWith: { a, _ in a })
        for id in Set(routes.keys).union(rules.keys) { apply(id, app: byID[id]) }
    }

    func setDevice(_ uid: String?, for bundleID: String) {
        let name = devices.first { $0.uid == uid }?.name
        update(bundleID) {
            $0.deviceUID = uid
            if let name { $0.deviceName = name }
        }
    }

    func setVolume(_ volume: Float, for bundleID: String) {
        update(bundleID) { $0.volume = volume }
    }

    func setLaunchAtLogin(_ on: Bool) {
        try? on ? SMAppService.mainApp.register() : SMAppService.mainApp.unregister()
        launchAtLogin = SMAppService.mainApp.status == .enabled
    }

    private func update(_ bundleID: String, _ change: (inout Rule) -> Void) {
        var rule = rule(for: bundleID)
        change(&rule)
        rules[bundleID] = rule.isDefault ? nil : rule
        store.save(rules)
        apply(bundleID, app: apps.first { $0.bundleID == bundleID })
    }

    /// Makes the live route for one app match its rule. No rule, app not running,
    /// or target device missing → no route (app plays on the system default). Rule is kept.
    private func apply(_ bundleID: String, app: AudioApp?) {
        let rule = rule(for: bundleID)
        guard let app, !rule.isDefault,
              let target = rule.deviceUID ?? defaultOutputUID(),
              devices.contains(where: { $0.uid == target }) else {
            routes[bundleID] = nil
            failed[bundleID] = nil
            return
        }
        if let route = routes[bundleID], route.deviceUID == target, Set(route.processes) == Set(app.processes) {
            route.volume = rule.volume
            return
        }
        routes[bundleID] = nil
        do {
            routes[bundleID] = try AppRoute(processes: app.processes, deviceUID: target, volume: rule.volume)
            failed[bundleID] = nil
        } catch {
            failed[bundleID] = "\(error)"
        }
    }
}

// MARK: - Audio capture permission

enum Permission { case granted, denied, unknown }

/// Reads the "System Audio Recording" TCC state. There is no public API, so this uses TCC's preflight
/// (same approach as Apple's sample-adjacent tools); unknown means "not asked yet" — the first tap asks.
func audioCapturePermission() -> Permission {
    typealias Preflight = @convention(c) (CFString, CFDictionary?) -> Int
    guard let handle = dlopen("/System/Library/PrivateFrameworks/TCC.framework/Versions/A/TCC", RTLD_NOW),
          let symbol = dlsym(handle, "TCCAccessPreflight") else { return .unknown }
    switch unsafeBitCast(symbol, to: Preflight.self)("kTCCServiceAudioCapture" as CFString, nil) {
    case 0: return .granted
    case 1: return .denied
    default: return .unknown
    }
}

```bash
#!/usr/bin/env bash
# Build build/soundctl.app. Signs with the first "Apple Development" identity so the
# audio capture permission survives rebuilds; falls back to ad-hoc (re-prompts each build).
#   ./Scripts/make-app.sh && open build/soundctl.app
set -euo pipefail
cd "$(dirname "$0")/.."

swift build -c release
BIN="$(swift build -c release --show-bin-path)/Soundctl"
APP="build/soundctl.app"

rm -rf "$APP"
mkdir -p "$APP/Contents/MacOS"
cp "$BIN" "$APP/Contents/MacOS/Soundctl"
cp Scripts/Info.plist "$APP/Contents/Info.plist"

IDENTITY="$(security find-identity -p codesigning -v | awk -F'"' '/Apple Development/ {print $2; exit}')"
codesign --force --timestamp=none --sign "${IDENTITY:--}" "$APP"
echo "built $APP (signed: ${IDENTITY:-ad-hoc})"
```
